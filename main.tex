\begin{frame}{Conglomerados como Categoría - Resumen}
\begin{itemize}
    \item Los objetos básicamente son los espacios Afines generalizados $A$ (punto, recta, plano, hiperplano, etc.), tal que su dimensión es $\delta(A) \in \Z$.
    \item Así $\delta(\textit{punto})=0$, $\delta(\textit{recta})=1$, etc.
    \item Existe una relación $u$ de inclusión generalizada entre dos objetos (un punto está en una recta, una recta está en un hiperplano, etc.).
    \item Sean $A,B$ objetos (espacios), son iguales si $A u B$ y $B u A$.
    \item Sean $A,B$ objetos (espacios), el espacio formado (span) por $A$ y $B$ es $A \uparrow B$ y el espacio coincidente (intersección) de $A$ y $B$ es $A \downarrow B$ (cuando exista).
    \item En $\Z$ se cumple:
    \[ \delta (A \uparrow B) = \delta(A) + \delta(B) - \delta(A \downarrow B) \]
 \end{itemize}
	
\end{frame}

\begin{frame}{Conglomerados como Categoría - Resumen}
\begin{itemize}
    \item Los morfismos básicamente se obtienen de $u$ entre espacios Afines generalizados $A$,$B$ (punto, recta, plano, hiperplano, etc.), dado que $u$ es una relación reflexiva, transitiva y antisimétrica.
    $\func{f}{A}{B} \Longleftrightarrow AuB$
    \item Así se pueden considerar los morfismos similares a los de una Categoría de POSets u órdenes parciales.
    \item Se cumple la clausura de morfismos por la composición y la existencia de identidades.
    \item De esta manera cada Conglomerado es una Categoría.
    \item \textit{Obs}: Se pueden analizar las propiedades de manera similar a la Categoría de Reales con la relación menor igual: $(\R,\le)$.
 \end{itemize}
\end{frame}

\begin{columns}
\column{0.5\textwidth}

\begin{itemize}
    \item 	Si $A,B \in [0]$ tales que $A$ y $B$ son distintos, entonces $(\exists! R \in [1])(AtR \wedge BtR)$, donde $R$ es $(A \uparrow B)$, los puntos $A$ y $B$ determinan la única recta $R$
    \item Si $A,B,C \in [0]$ distintos entre sí tales que $AtR, BtR, CtR$, con $R \cong (A \uparrow B)$ y $R \cong (A \uparrow C)$ entonces $(B \uparrow C) \cong R$.
    \item Sean $A,B,C \in [0]$ distintos entre sí tales que $AuK, BuK, CuK$ y $(A \uparrow B) \ncong (A \uparrow C)$ entonces $K \cong ((A \uparrow B) \uparrow(A \uparrow C))$
\end{itemize}
